Oproti v4.3 (chart/figure) pribúda **9 nových voliteľných šablón** — všetky prvky navrhnuté v ukážkovej prezentácii sú teraz natrvalo v engine ako pomenované `template` hodnoty. ## Static aj Dynamic engine — JS doplnok (renderers + buildSlide, nahraď existujúce)

```javascript
function donutSVG(percent, color){
    const c = 282.7, off = c - (percent/100)*c;
    return `<svg viewBox="0 0 120 120" width="110" height="110"><circle cx="60" cy="60" r="45" fill="none" stroke="#e2e8f0" stroke-width="14"/><circle cx="60" cy="60" r="45" fill="none" stroke="${color}" stroke-width="14" stroke-linecap="round" stroke-dasharray="${c}" stroke-dashoffset="${off}" transform="rotate(-90 60 60)"/><text x="60" y="67" text-anchor="middle" font-size="22" font-weight="800" fill="${color}">${percent}%</text></svg>`;
}
function gaugeSVG(percent, color){
    const r=50, cx=60, cy=60, ang=Math.PI*(1-percent/100);
    const x=(cx+r*Math.cos(ang)).toFixed(1), y=(cy-r*Math.sin(ang)).toFixed(1);
    return `<svg viewBox="0 0 120 70" width="160" height="95"><path d="M10 60 A50 50 0 0 1 110 60" fill="none" stroke="#e2e8f0" stroke-width="12"/><path d="M10 60 A50 50 0 0 1 ${x} ${y}" fill="none" stroke="${color}" stroke-width="12" stroke-linecap="round"/><text x="60" y="55" text-anchor="middle" font-size="20" font-weight="800" fill="${color}">${percent}%</text></svg>`;
}
const checkIcon = `<svg class="check-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><circle cx="12" cy="12" r="10"/><path d="m8 12 3 3 5-6"/></svg>`;
const quoteIcon = `<svg class="quote-mark" viewBox="0 0 24 24" fill="currentColor"><path d="M7 7c-2.5 0-4.5 2-4.5 4.5S4.5 16 7 16c.3 0 .6 0 .9-.1C7.3 18 5.7 19.5 4 20l.6 1.8C8.5 20.8 11 17.6 11 13.5 11 10 9.2 7 7 7zm10 0c-2.5 0-4.5 2-4.5 4.5s2 4.5 4.5 4.5c.3 0 .6 0 .9-.1-.6 2.1-2.2 3.6-3.9 4.1l.6 1.8c3.9-1 6.4-4.2 6.4-8.3 0-3.5-1.8-6.5-4-6.5z"/></svg>`;

// pridaj tieto kľúče do existujúceho 'renderers' objektu (vedľa grid-4/grid-3/.../list):
Object.assign(renderers, {
    "bignum": s => `<div class="content bignum">${s.boxes.map(b => `<div class="bignum-card"><div><span class="bignum-value">${b.value}<span class="bignum-unit">${b.unit||''}</span></span></div><div class="bignum-label">${b.label}</div></div>`).join('')}</div>`,
    "donut": s => `<div class="content donut">${s.boxes.map(b => `<div class="dg-card">${donutSVG(b.percent, b.color==='auto'||!b.color?'var(--secondary)':b.color)}<span class="box-title">${b.title}</span><div class="text-content">${b.desc||''}</div></div>`).join('')}</div>`,
    "gauge": s => `<div class="content gauge">${s.boxes.map(b => `<div class="dg-card">${gaugeSVG(b.percent, b.color==='auto'||!b.color?'var(--secondary)':b.color)}<span class="box-title">${b.title}</span><div class="text-content">${b.desc||''}</div></div>`).join('')}</div>`,
    "timeline": s => `<div class="content timeline-wrap"><div class="timeline-track">${s.items.map(it => `<div class="t-step"><div class="t-dot"></div><div class="t-label">${it.step}</div><div class="t-text">${it.text}</div></div>`).join('')}</div></div>`,
    "quote": s => `<div class="content quote-wrap"><div class="quote-box">${quoteIcon}<div class="quote-text">${s.text}</div><div class="quote-author">${s.author}</div></div></div>`,
    "checklist": s => `<div class="content">${s.pills ? `<div class="pill-row">${s.pills.map(p => `<span class="pill">${p}</span>`).join('')}</div>` : ''}<div class="check-list">${s.items.map(t => `<div class="check-item">${checkIcon}<span class="check-text">${t}</span></div>`).join('')}</div></div>`,
    "steps": s => `<div class="content steps-wrap">${s.items.map((it,i) => `<div class="step-row"><div class="step-num">${i+1}</div><div class="step-body"><div class="step-title">${it.title}</div><div class="step-text">${it.text}</div></div></div>`).join('')}</div>`,
    "overlap": s => `${s.pattern ? `<svg class="pattern-bg" viewBox="0 0 400 300" preserveAspectRatio="none"><defs><pattern id="dots" width="18" height="18" patternUnits="userSpaceOnUse"><circle cx="2" cy="2" r="1.4" fill="var(--secondary)" opacity=".18"/></pattern></defs><rect width="400" height="300" fill="url(#dots)"/></svg>` : ''}<div class="content overlap-wrap">${s.boxes.map(b => `<div class="overlap-card"><span class="box-title">${b.t}</span><div class="text-content">${b.d}</div></div>`).join('')}</div>`
});

// v renderer pre 'single-block' priprav dropcap modifikátor (nahraď existujúci riadok):
renderers["single-block"] = s => `<div class="content single-block"><div class="single-card${s.dropcap ? ' dropcap' : ''}"><span class="box-title">${s.heading}</span><div class="text-content">${s.body}</div>${s.figure ? `<div class="graph-box fig">${s.figure}</div>` : ''}</div></div>`;
```

`buildSlide()` (Static aj Dynamic) sa nemení — už volá `renderers[s.template](s)`, takže nové kľúče fungujú automaticky, len ich treba pridať do `renderers` objektom vyššie (`Object.assign`).

## Zhrnutie — čo si treba vybrať

Všetkých 9 prvkov je teraz **v engine k dispozícii**, ale AI ich generuje len keď to `when_to_use` naozaj odôvodňuje. Nič sa aktívne "nezapína" plošne — ak chceš, aby sa niektorý prvok objavoval **častejšie/preferenčne** (napr. vždy uprednostniť `donut` pred bežnou grid kartou pri percentách), stačí to pridať ako samostatné pravidlo do `presentation_structure_rules`.
