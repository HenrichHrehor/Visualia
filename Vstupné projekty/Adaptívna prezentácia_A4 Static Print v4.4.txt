<script>
const AUTHOR_DEFAULT = "Autor prezentácie";

/* =========================================================================
   DÁTA PREZENTÁCIE (Generované AI)
   ========================================================================= */
const data = [
    // Sem vkladá model pole objektov pre jednotlivé slajdy
];

function card(b) { 
    const iconHtml = b.icon ? `<div class="card-icon">${b.icon}</div>` : '';
    const chartHtml = b.chart ? `<div class="graph-box">${b.chart}</div>` : '';
    return `<div class="card">
        <div>
            <div class="card-header">${iconHtml}<span class="box-title">${b.t}</span></div>
            <div class="text-content">${b.d}</div>
        </div>
        ${chartHtml}
    </div>`; 
}

function listItem(it, i) { 
    return `<div class="list-item"><div class="list-marker">${i + 1}</div><div class="list-body"><span class="box-title">${it.t}</span><div class="text-content">${it.d}</div></div></div>`; 
}

const renderers = {
    "grid-4": s => `<div class="content grid-cards-4">${s.boxes.map(card).join('')}</div>`,
    "grid-3": s => `<div class="content grid-cards-3" style="${s.extraStyle || ''}">${s.boxes.map(card).join('')}</div>`,
    "grid-2-airy": s => `<div class="content grid-cards-2">${s.boxes.map(card).join('')}</div>`,
    "grid-1": s => `<div class="content grid-cards-1">${s.boxes.map(card).join('')}</div>`,
    "single-block": s => `<div class="content single-block"><div class="single-card"><span class="box-title">${s.heading}</span><div class="text-content">${s.body}</div>${s.figure ? `<div class="graph-box fig">${s.figure}</div>` : ''}</div></div>`,
    "list": s => `<div class="content list-block">${s.items.map(listItem).join('')}</div>`
};

function buildSlide(s, index, total) {
    const themeClass = `theme-${s.theme}`;
    const authorText = s.author || AUTHOR_DEFAULT;
    
    if (s.template === "hero") {
        return `<div class="slide ${themeClass} hero">
            <h1 class="hero-title">${s.title}</h1>
            ${s.subtitle ? `<p class="hero-subtitle">${s.subtitle}</p>` : ''}
            <div class="hero-author"><strong>${s.author}</strong></div>
            <div class="footer"><span style="width:100%; text-align:center;">Snímka ${index + 1} / ${total}</span></div>
        </div>`;
    }
    const headerHtml = `<div class="header">${s.tag ? `<div class="header-tag">${s.tag}</div>` : ''}<h1 class="title">${s.title || ''}</h1></div>`;
    const contentHtml = renderers[s.template](s);
    const footerHtml = `<div class="footer"><span>${authorText}</span><span>Snímka ${index + 1} / ${total}</span></div>`;
    return `<div class="slide ${themeClass}">${headerHtml}${contentHtml}${footerHtml}</div>`;
}

document.getElementById('slides').innerHTML = data.map((slide, index) => buildSlide(slide, index, data.length)).join('');
</script>
